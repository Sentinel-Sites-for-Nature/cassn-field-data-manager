# CA-SSN Field Data Manager — Python dependencies
# Install with: pip install -r requirements.txt
#
# NOTE: PyExifTool is a wrapper around the `exiftool` command-line tool, which
# must be installed separately:
#   macOS:   brew install exiftool
#   Windows: download from https://exiftool.org and add it to PATH

box-sdk-gen      # Box cloud storage + OAuth
Pillow           # image EXIF reading
piexif           # image EXIF reading
PyExifTool       # camera metadata (requires the exiftool binary, see note above)
